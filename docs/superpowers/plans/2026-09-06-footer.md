- [ ] Implement this complete render structure. The Privacy Policy URL is internal to the production site, so it intentionally stays in the same tab and does not receive an arrow despite being absolute in the static prototype:

```js
function Footer(){
  const footerGroups=[
    {title:"Navigate",links:[
      {label:"What We Do",href:"#what-we-do"},
      {label:"Events",href:"#events"},
      {label:"Team",href:"#team"},
      {label:"Partners",href:"#partners"},
      {label:"FAQ",href:"#faq"}
    ]},
    {title:"Resources",links:[
      {label:"Global Hackathon",href:"https://de.superteam.fun/buildstation",external:true},
      {label:"Resources & Docs",href:"https://superteamde.gitbook.io/superteamde",external:true},
      {label:"Superteam Global",href:"https://superteam.fun",external:true}
    ]},
    {title:"Connect",links:[
      {label:"X",href:"https://x.com/SuperteamDE",external:true},
      {label:"Discord",href:"https://discord.gg/CVwJhHgFfF",external:true},
      {label:"Telegram",href:"https://t.me/superteamgermany",external:true},
      {label:"YouTube",href:"https://www.youtube.com/@SuperteamDE",external:true}
    ]}
  ];
  return e("footer",{className:"site-footer",id:"footer","aria-labelledby":"footer-heading"},
    e("h2",{className:"site-footer-sr",id:"footer-heading"},"Footer"),
    e("div",{className:"site-footer-inner"},
      e("div",{className:"site-footer-main"},
        e("div",{className:"site-footer-brand"},
          e("a",{href:"#",className:"site-footer-home","aria-label":"Back to the top"},e("img",{src:"https://de.superteam.fun/images/stLogoWithIcon.svg",alt:"Superteam Germany"})),
          e("p",{className:"site-footer-tagline"},"Building Solana’s future in Germany.")
        ),
        ...footerGroups.map(group=>e("nav",{className:"site-footer-group",key:group.title,"aria-label":group.title},
          e("h3",null,group.title),
          e("ul",{className:"site-footer-links"},...group.links.map(link=>e("li",{key:link.label},
            e("a",{className:"site-footer-link",href:link.href,target:link.external?"_blank":undefined,rel:link.external?"noreferrer":undefined},link.label,link.external?e("span",{className:"site-footer-arrow","aria-hidden":"true"},"↗"):null)
          )))
        ))
      ),
      e("div",{className:"site-footer-legal"},
        e("p",null,"© 2026 Superteam Germany. All rights reserved."),
        e("a",{className:"site-footer-privacy",href:"https://de.superteam.fun/policy"},"Privacy Policy")
      )
    )
  );
}
```
